## execise 2

```js
app.get("/products", async (req, res) => {
  try {
    const stringifyProducts = await fs.readFile(
      `${__dirname}/db/products.json`,
    );
    const products = JSON.parse(stringifyProducts);
    res.json(products);
  } catch (error) {
    console.error({
      message: "Failed to fetch products",
      endpoint: "/products",
      errorMsg: error.message, //err-->>error הבאג היה כאן
    });
    res.status(500).send("Internal server error");
  }
});
```

- לולי תיקון השגיאה, אם התוכנה תיכנס ל-catch, יהיה עוד error, כי התוכנה לא מכירה אובייקט בשם err, וכן הלקוח לא יקבל את ההודעה "Internal server error"

## execise 3

```js
document.getElementById("load-btn").addEventListener("click", async (e) => {
  try {
    const leaguesRes = await fetch("http://localhost:3000/leagues", {
      method: "GET",
      headers: { "Content-Type": "application/json" },
    });

    const leagues = await leaguesRes.json();

    const teamRequests = [];
    for (const league of leagues) {
      teamRequests.push(
        fetch("http://localhost:3000/bulk-teams", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.strigify(leagues.teams),
        }),
      );
    }

    const teamHeaders = await Promise.all(teamRequests);

    const teamBodyRequests = [];
    for (const header of teamHeaders) {
      teamBodyRequests.push(header.json());
    }

    const teamsResponse = await Promise.all(teamBodyRequests);

    console.log(teamsResponse);
  } catch (error) {
    console.error({ data: error.message });
  }
});
```

## execise 4

```js
app.patch("/product/:id", async (req, res) => {
  try {
    const productId = req.params.id;
    const stringifyProduct = await fs.readFile(`${__dirname}/db/products.json`);

    const products = JSON.parse(stringifyProduct);

    const product = products.find((product) => product.id == productId);
    if (!product) {
      return res.status(404).send("Product not found");
    }

    const newProducts = products.map((item) =>
      item.id == product.id
        ? {
            ...product,
            ...req.body,
          }
        : item,
    );
    await fs.writeFile(
       `${__dirname}/db/products.json`,
       JSON.stringify(newProducts, null, 2),
    );
    res.json(newProducts);
  } catch (error) {
    res.status(500).send("Internal server error");
  }
});
```
